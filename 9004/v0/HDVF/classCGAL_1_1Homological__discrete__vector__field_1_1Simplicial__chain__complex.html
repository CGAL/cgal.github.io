template<typename CoefficientRing, typename Traits, typename SparseMatrixStruct = OSM::Sparse_matrix_structure<OSM::Sparse_chain>>
class CGAL::Homological_discrete_vector_field::Simplicial_chain_complex< CoefficientRing, Traits, SparseMatrixStruct >

The class Simplicial_chain_complex refines the Abstract_simplicial_chain_complex class by assigning coordinates to vertices (i.e. 0-simplices).

Hence, vtk output is available.
